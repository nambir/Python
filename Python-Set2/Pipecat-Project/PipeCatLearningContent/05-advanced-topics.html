# Optimized audio processing configuration
import asyncio
from pipecat import Pipeline
from pipecat.transports import LocalTransport
from pipecat.processors.voice import OpenAISTTProcessor, OpenAITTSProcessor
from pipecat.processors.llm import OpenAILLMProcessor

class OptimizedVoiceAI:
    def __init__(self):
        self.setup_optimized_processors()
    
    def setup_optimized_processors(self):
        """Configure processors for optimal performance"""
        # Optimized STT with lower latency
        self.stt = OpenAISTTProcessor(
            api_key=os.getenv("OPENAI_API_KEY"),
            model="whisper-1",
            language="en",
            temperature=0.0,  # Deterministic output
            response_format="json"  # Faster parsing
        )
        
        # Optimized LLM with streaming
        self.llm = OpenAILLMProcessor(
            api_key=os.getenv("OPENAI_API_KEY"),
            model="gpt-4",
            temperature=0.7,
            max_tokens=150,  # Limit response length
            stream=True  # Enable streaming for faster responses
        )
        
        # Optimized TTS with faster voice
        self.tts = OpenAITTSProcessor(
            api_key=os.getenv("OPENAI_API_KEY"),
            model="tts-1",  # Faster model
            voice="alloy",
            speed=1.1  # Slightly faster speech
        )
    
    async def run_optimized(self):
        """Run with performance optimizations"""
        transport = LocalTransport()
        
        # Build optimized pipeline
        pipeline = Pipeline([
            transport.input(),
            self.stt,
            self.llm,
            self.tts,
            transport.output()
        ])
        
        # Run with performance monitoring
        await self.run_with_monitoring(pipeline)
    
    async def run_with_monitoring(self, pipeline):
        """Run pipeline with performance monitoring"""
        start_time = asyncio.get_event_loop().time()
        
        try:
            await pipeline.run()
        finally:
            end_time = asyncio.get_event_loop().time()
            print(f"Pipeline runtime: {end_time - start_time:.2f} seconds")

Memory and Resource Management

# Memory-efficient voice AI implementation
import gc
import psutil
import asyncio
from contextlib import asynccontextmanager

class MemoryEfficientVoiceAI:
    def __init__(self):
        self.memory_threshold = 512 * 1024 * 1024  # 512MB
        self.setup_memory_monitoring()
    
    def setup_memory_monitoring(self):
        """Set up memory monitoring"""
        self.memory_check_interval = 30  # Check every 30 seconds
        self.memory_monitor_task = None
    
    async def start_memory_monitoring(self):
        """Start memory monitoring task"""
        self.memory_monitor_task = asyncio.create_task(self.monitor_memory())
    
    async def monitor_memory(self):
        """Monitor memory usage and cleanup if needed"""
        while True:
            try:
                memory_usage = psutil.Process().memory_info().rss
                if memory_usage > self.memory_threshold:
                    print(f"High memory usage detected: {memory_usage / 1024 / 1024:.2f}MB")
                    await self.cleanup_memory()
                await asyncio.sleep(self.memory_check_interval)
            except Exception as e:
                print(f"Memory monitoring error: {e}")
                break
    
    async def cleanup_memory(self):
        """Clean up memory"""
        gc.collect()  # Force garbage collection
        print("Memory cleanup performed")
    
    @asynccontextmanager
    async def managed_session(self):
        """Context manager for resource management"""
        try:
            await self.start_memory_monitoring()
            yield self
        finally:
            if self.memory_monitor_task:
                self.memory_monitor_task.cancel()
            await self.cleanup_memory()

# Comprehensive monitoring setup
import asyncio
import time
from dataclasses import dataclass
from typing import Dict, Any
import json

@dataclass
class Metrics:
    """Application metrics"""
    total_requests: int = 0
    successful_requests: int = 0
    failed_requests: int = 0
    average_response_time: float = 0.0
    active_sessions: int = 0
    memory_usage: float = 0.0
    cpu_usage: float = 0.0

class VoiceAIMonitor:
    def __init__(self):
        self.metrics = Metrics()
        self.start_time = time.time()
        self.request_times = []
    
    def record_request(self, response_time: float, success: bool):
        """Record request metrics"""
        self.metrics.total_requests += 1
        if success:
            self.metrics.successful_requests += 1
        else:
            self.metrics.failed_requests += 1
        
        self.request_times.append(response_time)
        if len(self.request_times) > 100:  # Keep last 100 requests
            self.request_times.pop(0)
        
        self.metrics.average_response_time = sum(self.request_times) / len(self.request_times)
    
    def get_uptime(self) -> float:
        """Get application uptime in seconds"""
        return time.time() - self.start_time
    
    def get_health_status(self) -> Dict[str, Any]:
        """Get application health status"""
        return {
            "status": "healthy" if self.metrics.failed_requests / max(1, self.metrics.total_requests) < 0.1 else "degraded",
            "uptime": self.get_uptime(),
            "metrics": {
                "total_requests": self.metrics.total_requests,
                "successful_requests": self.metrics.successful_requests,
                "failed_requests": self.metrics.failed_requests,
                "average_response_time": self.metrics.average_response_time,
                "active_sessions": self.metrics.active_sessions
            }
        }
    
    async def start_monitoring(self):
        """Start monitoring task"""
        while True:
            try:
                # Update system metrics
                import psutil
                self.metrics.memory_usage = psutil.Process().memory_info().rss / 1024 / 1024  # MB
                self.metrics.cpu_usage = psutil.Process().cpu_percent()
                
                # Log metrics
                print(f"Metrics: {json.dumps(self.get_health_status(), indent=2)}")
                
                await asyncio.sleep(60)  # Update every minute
            except Exception as e:
                print(f"Monitoring error: {e}")
                await asyncio.sleep(60)

Error Handling and Recovery

# Robust error handling and recovery
import asyncio
import logging
from typing import Optional, Callable
from functools import wraps

class VoiceAIErrorHandler:
    def __init__(self, max_retries: int = 3, retry_delay: float = 1.0):
        self.max_retries = max_retries
        self.retry_delay = retry_delay
        self.logger = logging.getLogger(__name__)
    
    def retry_on_failure(self, func: Callable):
        """Decorator for retrying failed operations"""
        @wraps(func)
        async def wrapper(*args, **kwargs):
            last_exception = None
            
            for attempt in range(self.max_retries + 1):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < self.max_retries:
                        self.logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {self.retry_delay}s...")
                        await asyncio.sleep(self.retry_delay)
                    else:
                        self.logger.error(f"All {self.max_retries + 1} attempts failed. Last error: {e}")
            
            raise last_exception
        
        return wrapper
    
    async def handle_audio_error(self, error: Exception) -> str:
        """Handle audio processing errors"""
        if "microphone" in str(error).lower():
            return "I'm having trouble hearing you. Please check your microphone and try again."
        elif "speaker" in str(error).lower():
            return "I'm having trouble speaking. Please check your speakers and try again."
        else:
            return "I encountered an audio processing error. Please try again."
    
    async def handle_api_error(self, error: Exception) -> str:
        """Handle API errors"""
        if "rate limit" in str(error).lower():
            return "I'm experiencing high demand. Please wait a moment and try again."
        elif "authentication" in str(error).lower():
            return "I'm having trouble connecting to my services. Please try again later."
        else:
            return "I encountered a service error. Please try again."

# Load balancer configuration for scaling
from fastapi import FastAPI, WebSocket
from fastapi.middleware.cors import CORSMiddleware
import asyncio
import json
from typing import Dict, List

class VoiceAILoadBalancer:
    def __init__(self, max_instances: int = 10):
        self.max_instances = max_instances
        self.active_instances: List[VoiceAIInstance] = []
        self.instance_weights: Dict[int, int] = {}
        self.app = FastAPI()
        self.setup_middleware()
        self.setup_routes()
    
    def setup_middleware(self):
        """Set up CORS and other middleware"""
        self.app.add_middleware(
            CORSMiddleware,
            allow_origins=["*"],
            allow_credentials=True,
            allow_methods=["*"],
            allow_headers=["*"],
        )
    
    def setup_routes(self):
        """Set up API routes"""
        @self.app.websocket("/ws")
        async def websocket_endpoint(websocket: WebSocket):
            await websocket.accept()
            instance = await self.get_available_instance()
            if instance:
                await instance.handle_connection(websocket)
            else:
                await websocket.send_text(json.dumps({"error": "No available instances"}))
                await websocket.close()
        
        @self.app.get("/health")
        async def health_check():
            return {
                "status": "healthy",
                "active_instances": len(self.active_instances),
                "max_instances": self.max_instances
            }
    
    async def get_available_instance(self) -> Optional[VoiceAIInstance]:
        """Get an available voice AI instance"""
        for instance in self.active_instances:
            if instance.is_available():
                return instance
        
        if len(self.active_instances) < self.max_instances:
            new_instance = VoiceAIInstance()
            await new_instance.initialize()
            self.active_instances.append(new_instance)
            return new_instance
        
        return None
    
    async def scale_up(self):
        """Scale up by adding more instances"""
        if len(self.active_instances) < self.max_instances:
            new_instance = VoiceAIInstance()
            await new_instance.initialize()
            self.active_instances.append(new_instance)
            print(f"Scaled up to {len(self.active_instances)} instances")
    
    async def scale_down(self):
        """Scale down by removing unused instances"""
        if len(self.active_instances) > 1:
            # Remove the least used instance
            instance = min(self.active_instances, key=lambda x: x.get_usage_count())
            await instance.cleanup()
            self.active_instances.remove(instance)
            print(f"Scaled down to {len(self.active_instances)} instances")
