📊 Variables and Data Types

Basic Data Types

# Numbers
age = 25                    # Integer
temperature = 98.6          # Float
pi = 3.14159               # Float

# Text
name = "Alice"             # String
message = 'Hello World!'   # String (single quotes work too)

# Boolean (True/False)
is_learning = True
has_experience = False

# Print variables
print(f"Name: {name}, Age: {age}")
print(f"Temperature: {temperature}°F")

Lists and Dictionaries

# Lists (ordered collections)
fruits = ["apple", "banana", "orange"]
numbers = [1, 2, 3, 4, 5]
mixed = [1, "hello", True, 3.14]

# Accessing list elements
print(fruits[0])           # "apple"
print(fruits[-1])          # "orange" (last element)

# Dictionaries (key-value pairs)
person = {
    "name": "Alice",
    "age": 25,
    "city": "New York",
    "is_student": True
}

# Accessing dictionary values
print(person["name"])      # "Alice"
print(person.get("age"))   # 25 (safer method)

🔧 Functions

Creating Functions

# Basic function
def greet(name):
    return f"Hello, {name}!"

# Function with multiple parameters
def calculate_area(length, width):
    area = length * width
    return area

# Function with default parameters
def greet_with_title(name, title="Mr./Ms."):
    return f"Hello, {title} {name}!"

# Using functions
message = greet("Alice")
print(message)  # "Hello, Alice!"

room_area = calculate_area(10, 5)
print(f"Room area: {room_area} square feet")

formal_greeting = greet_with_title("Smith", "Dr.")
casual_greeting = greet_with_title("Smith")  # Uses default title

Lambda Functions (Quick Functions)

# Lambda function for simple operations
square = lambda x: x * x
add = lambda a, b: a + b

# Using lambda functions
print(square(5))    # 25
print(add(3, 4))    # 7

# Lambda with lists
numbers = [1, 2, 3, 4, 5]
squared_numbers = list(map(lambda x: x * x, numbers))
print(squared_numbers)  # [1, 4, 9, 16, 25]

⚠️ Error Handling

Try-Except Blocks

# Basic error handling
try:
    age = int(input("Enter your age: "))
    print(f"You are {age} years old")
except ValueError:
    print("Please enter a valid number")

# Multiple exception types
try:
    file = open("data.txt", "r")
    content = file.read()
    file.close()
except FileNotFoundError:
    print("File not found")
except PermissionError:
    print("Permission denied")
except Exception as e:
    print(f"An error occurred: {e}")

# Finally block (always executes)
try:
    result = 10 / 0
except ZeroDivisionError:
    print("Cannot divide by zero")
finally:
    print("This always runs")
⚠️ Important: Good error handling is crucial in AI applications. Users might say unexpected things, and we need to handle these gracefully!
