🎯 What is Model Context Protocol (MCP)?

Model Context Protocol (MCP) is a standardized protocol that allows AI models to connect with external tools, data sources, and services. It enables your voice AI to:

🔧 Access External Tools

Connect with APIs, databases, and third-party services

📊 Real-time Data

Access live information like weather, news, and stock prices

🎯 Function Calling

Execute specific functions based on user requests

🌐 Web Integration

Interact with web services and online platforms

💡 Key Benefit: MCP transforms your voice AI from a simple chatbot into a powerful assistant that can perform real-world tasks!

🔧 Integrating MCP with PipecatAI

Setting Up MCP Client

# Enhanced voice AI with MCP integration
import asyncio
import os
from dotenv import load_dotenv
from pipecat import Pipeline
from pipecat.transports import LocalTransport
from pipecat.processors.voice import OpenAISTTProcessor, OpenAITTSProcessor
from pipecat.processors.llm import OpenAILLMProcessor
from pipecat.processors.aggregators import LLMAssistantResponseAggregator
from mcp import Client
from mcp.types import Tool

load_dotenv()

class MCPVoiceAI:
    def __init__(self):
        self.api_key = os.getenv("OPENAI_API_KEY")
        self.mcp_client = Client()
        self.setup_tools()
    
    def setup_tools(self):
        """Set up MCP tools for the AI to use"""
        self.tools = [
            Tool(
                name="get_weather",
                description="Get current weather information for a location",
                inputSchema={
                    "type": "object",
                    "properties": {
                        "location": {"type": "string", "description": "City name or location"}
                    },
                    "required": ["location"]
                }
            ),
            Tool(
                name="search_web",
                description="Search the web for information",
                inputSchema={
                    "type": "object",
                    "properties": {
                        "query": {"type": "string", "description": "Search query"}
                    },
                    "required": ["query"]
                }
            ),
            Tool(
                name="calculate",
                description="Perform mathematical calculations",
                inputSchema={
                    "type": "object",
                    "properties": {
                        "expression": {"type": "string", "description": "Math expression to calculate"}
                    },
                    "required": ["expression"]
                }
            )
        ]
    
    async def run(self):
        """Run the enhanced voice AI with MCP integration"""
        # Create processors
        stt = OpenAISTTProcessor(api_key=self.api_key)
        llm = OpenAILLMProcessor(
            api_key=self.api_key,
            model="gpt-4",
            tools=self.tools,  # Pass tools to LLM
            system_message="You are a helpful AI assistant with access to various tools. Use them when appropriate to help users."
        )
        tts = OpenAITTSProcessor(api_key=self.api_key)
        
        # Create transport and aggregator
        transport = LocalTransport()
        aggregator = LLMAssistantResponseAggregator()
        
        # Build pipeline
        pipeline = Pipeline([
            transport.input(),
            stt,
            llm,
            aggregator,
            tts,
            transport.output()
        ])
        
        print("🎤 Enhanced Voice AI with MCP is ready!")
        print("Available tools: Weather, Web Search, Calculator")
        print("Press Ctrl+C to stop")
        
        try:
            await pipeline.run()
        except KeyboardInterrupt:
            print("\n👋 Voice AI stopped. Goodbye!")

# Run the enhanced voice AI
if __name__ == "__main__":
    voice_ai = MCPVoiceAI()
    asyncio.run(voice_ai.run())

Implementing MCP Tools

# MCP Tool Implementations
import requests
import json
from mcp.types import Tool

class WeatherTool:
    def __init__(self, api_key):
        self.api_key = api_key
        self.base_url = "http://api.openweathermap.org/data/2.5/weather"
    
    async def get_weather(self, location: str) -> str:
        """Get weather information for a location"""
        try:
            params = {
                "q": location,
                "appid": self.api_key,
                "units": "metric"
            }
            response = requests.get(self.base_url, params=params)
            data = response.json()
            
            if response.status_code == 200:
                weather = data["weather"][0]["description"]
                temp = data["main"]["temp"]
                humidity = data["main"]["humidity"]
                
                return f"Weather in {location}: {weather}, Temperature: {temp}°C, Humidity: {humidity}%"
            else:
                return f"Sorry, I couldn't get weather information for {location}"
        except Exception as e:
            return f"Error getting weather: {str(e)}"

class WebSearchTool:
    def __init__(self, api_key):
        self.api_key = api_key
        self.base_url = "https://api.bing.microsoft.com/v7.0/search"
    
    async def search_web(self, query: str) -> str:
        """Search the web for information"""
        try:
            headers = {"Ocp-Apim-Subscription-Key": self.api_key}
            params = {"q": query, "count": 5}
            
            response = requests.get(self.base_url, headers=headers, params=params)
            data = response.json()
            
            if response.status_code == 200:
                results = data["webPages"]["value"]
                summary = "Here are the top search results:\n"
                for i, result in enumerate(results[:3], 1):
                    summary += f"{i}. {result['name']}: {result['snippet']}\n"
                return summary
            else:
                return f"Sorry, I couldn't search for '{query}'"
        except Exception as e:
            return f"Error searching: {str(e)}"

class CalculatorTool:
    async def calculate(self, expression: str) -> str:
        """Perform mathematical calculations"""
        try:
            # Simple calculator - in production, use a proper math parser
            result = eval(expression)
            return f"The result of {expression} is {result}"
        except Exception as e:
            return f"Error calculating '{expression}': {str(e)}"

Custom MCP Server

# Custom MCP Server for advanced integrations
from mcp import Server
from mcp.types import Tool, Resource
import asyncio

class CustomMCPServer:
    def __init__(self):
        self.server = Server()
        self.setup_resources()
        self.setup_tools()
    
    def setup_resources(self):
        """Set up MCP resources"""
        self.resources = [
            Resource(
                uri="file:///data/contacts.json",
                name="contacts",
                description="User contact information"
            ),
            Resource(
                uri="file:///data/calendar.json",
                name="calendar",
                description="User calendar events"
            )
        ]
    
    def setup_tools(self):
        """Set up custom tools"""
        self.tools = [
            Tool(
                name="add_contact",
                description="Add a new contact to the user's contact list",
                inputSchema={
                    "type": "object",
                    "properties": {
                        "name": {"type": "string"},
                        "email": {"type": "string"},
                        "phone": {"type": "string"}
                    },
                    "required": ["name", "email"]
                }
            ),
            Tool(
                name="schedule_meeting",
                description="Schedule a meeting in the user's calendar",
                inputSchema={
                    "type": "object",
                    "properties": {
                        "title": {"type": "string"},
                        "date": {"type": "string"},
                        "time": {"type": "string"},
                        "duration": {"type": "integer"}
                    },
                    "required": ["title", "date", "time"]
                }
            )
        ]
    
    async def handle_tool_call(self, tool_name: str, arguments: dict) -> str:
        """Handle tool execution"""
        if tool_name == "add_contact":
            return await self.add_contact(arguments)
        elif tool_name == "schedule_meeting":
            return await self.schedule_meeting(arguments)
        else:
            return f"Unknown tool: {tool_name}"
    
    async def add_contact(self, args: dict) -> str:
        """Add a contact to the user's contact list"""
        # Implementation for adding contact
        return f"Added contact: {args['name']} ({args['email']})"
    
    async def schedule_meeting(self, args: dict) -> str:
        """Schedule a meeting in the user's calendar"""
        # Implementation for scheduling meeting
        return f"Scheduled meeting: {args['title']} on {args['date']} at {args['time']}"

# Usage example
async def main():
    server = CustomMCPServer()
    # Start the MCP server
    await server.server.run()

if __name__ == "__main__":
    asyncio.run(main())

Real-time Data Integration

# Real-time data integration with MCP
import asyncio
import aiohttp
from datetime import datetime

class RealTimeDataTool:
    def __init__(self):
        self.session = None
    
    async def __aenter__(self):
        self.session = aiohttp.ClientSession()
        return self
    
    async def __aexit__(self, exc_type, exc_val, exc_tb):
        if self.session:
            await self.session.close()
    
    async def get_stock_price(self, symbol: str) -> str:
        """Get real-time stock price"""
        try:
            url = f"https://api.example.com/stock/{symbol}"
            async with self.session.get(url) as response:
                data = await response.json()
                price = data.get("price", "N/A")
                return f"Current price of {symbol}: ${price}"
        except Exception as e:
            return f"Error getting stock price: {str(e)}"
    
    async def get_crypto_price(self, symbol: str) -> str:
        """Get real-time cryptocurrency price"""
        try:
            url = f"https://api.coingecko.com/api/v3/simple/price?ids={symbol}&vs_currencies=usd"
            async with self.session.get(url) as response:
                data = await response.json()
                price = data.get(symbol, {}).get("usd", "N/A")
                return f"Current price of {symbol}: ${price}"
        except Exception as e:
            return f"Error getting crypto price: {str(e)}"
    
    async def get_news_headlines(self, category: str = "general") -> str:
        """Get latest news headlines"""
        try:
            url = f"https://newsapi.org/v2/top-headlines?category={category}&apiKey=YOUR_API_KEY"
            async with self.session.get(url) as response:
                data = await response.json()
                articles = data.get("articles", [])
                headlines = []
                for article in articles[:5]:
                    headlines.append(f"• {article['title']}")
                return "Latest news headlines:\n" + "\n".join(headlines)
        except Exception as e:
            return f"Error getting news: {str(e)}"
