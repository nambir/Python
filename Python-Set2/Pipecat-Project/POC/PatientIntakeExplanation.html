2. Data Collection Logic

class PatientIntakeProcessor(FrameProcessor):
    def __init__(self):
        self.patient_data = {
            "name": None,
            "dob": None,
            "symptoms": None,
            "medications": [],
            "allergies": []
        }
        self.current_step = "greeting"

    async def process_frame(self, frame, direction):
        if isinstance(frame, TranscriptionFrame):
            text = frame.text.lower()

            if self.current_step == "greeting":
                # Extract name from "Hi, I'm John Doe"
                name = self.extract_name(text)
                if name:
                    self.patient_data["name"] = name
                    self.current_step = "date_of_birth"
                    response = f"Nice to meet you, {name}. What's your date of birth?"
                else:
                    response = "I didn't catch your name. Could you please tell me your full name?"

            elif self.current_step == "date_of_birth":
                # Extract DOB from "I was born on March 15th, 1985"
                dob = self.extract_date(text)
                if dob:
                    self.patient_data["dob"] = dob
                    self.current_step = "symptoms"
                    response = "Thank you. What brings you in today? What symptoms are you experiencing?"
                else:
                    response = "I need your date of birth. Please say it like 'March 15th, 1985'"

            # Continue for other steps...

            await self.push_frame(TextFrame(text=response), direction)

3. Natural Language Processing

def extract_name(self, text):
    # Look for patterns like "I'm John", "My name is Sarah", etc.
    patterns = [
        r"i'm\s+([a-zA-Z\s]+)",
        r"my name is\s+([a-zA-Z\s]+)",
        r"this is\s+([a-zA-Z\s]+)"
    ]

    for pattern in patterns:
        match = re.search(pattern, text, re.IGNORECASE)
        if match:
            return match.group(1).strip().title()
    return None

def extract_symptoms(self, text):
    # Medical keyword detection
    symptom_keywords = {
        "pain": ["hurt", "ache", "pain", "sore"],
        "fever": ["fever", "hot", "temperature"],
        "cough": ["cough", "coughing"],
        "headache": ["headache", "head hurt"]
    }

    found_symptoms = []
    for symptom, keywords in symptom_keywords.items():
        if any(keyword in text.lower() for keyword in keywords):
            found_symptoms.append(symptom)

    return found_symptoms
